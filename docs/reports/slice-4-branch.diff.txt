+import { z } from 'zod';
+import { stripZeroWidth } from '../matcher.js';
+import { createWindowStore, DEFAULT_MAX_TRACKED_MEMBERS } from '../window.js';
+import type { AutomodRule, CompiledPredicate, Predicate } from './types.js';
+
+export const duplicateMessageConfig = z
+  .object({
+    /** Fires when the same normalised text is repeated more than this many times within `windowMs`. */
+    maxRepeats: z.number().int().min(1).max(1_000),
+    /** Bounded 1 s-10 min, the same reasoning as `rapid_spam`'s `windowMs`. */
+    windowMs: z.number().int().min(1_000).max(600_000),
+    /**
+     * Below this many normalised characters, the rule does not fire at all.
+     *
+     * **Defaults to 0 — no floor unless the operator sets one.** A non-zero
+     * default would be this plan quietly choosing a false-negative rate on the
+     * operator's behalf, in the tier whose gate is a false-positive rate (plan
+     * ruling, Task 4). With `minLength: 0`, `"ok"` posted three times *does*
+     * fire — a floor still matters for a real deployment, and an operator
+     * should set one, but the schema itself must not silently suppress a
+     * match an operator configured `maxRepeats`/`windowMs` to catch.
+     */
+    minLength: z.number().int().min(0).max(4_000).default(0),
+  })
+  .strict();
+
+export type DuplicateMessageConfig = z.infer<typeof duplicateMessageConfig>;
+
+/**
+ * `duplicate_message`'s own normalisation — **not** the keyword matcher's.
+ *
+ * The spec's instruction ("the normalisation is the same code the keyword
+ * matcher uses, not a second copy") is not achievable, for two verified
+ * reasons (plan ruling PR-20):
+ *
+ * 1. `matcher.ts`'s `collapseDuplicatesTransformer` stage takes its
+ *    thresholds from the *configured word list* (`collapseThresholdsFor`,
+ *    ADR 0039) — this predicate has no word list, so that stage has no
+ *    defined behaviour for it.
+ * 2. `obscenity` never exposes its transformer chain as a function over a
+ *    string: `TransformerSet` is internal to `dist/` and not part of the
+ *    package's public exports. Reusing the matcher's exact chain would mean a
+ *    deep import into `obscenity/dist`, which this project does not do.
+ *
+ * A third option was examined and rejected: `obscenity` *does* export the
+ * transformer factories themselves (`resolveConfusablesTransformer`,
+ * `resolveLeetSpeakTransformer`, `toAsciiLowerCaseTransformer`), so one could
+ * hand-roll the dispatch loop that runs them over a string. Rejected because
+ * the result would be fragile *and* still not faithful: it would depend on an
+ * undocumented container discriminant with no compatibility guarantee, and it
+ * would still omit `collapseDuplicatesTransformer` (reason 1 above) — a
+ * partial, brittle subset of the matcher's behaviour, worse than an honest,
+ * simple normaliser that says what it does.
+ *
+ * So this normalises with `stripZeroWidth` — the one function `matcher.ts`
+ * actually exports, imported rather than reimplemented, so the one evasion
+ * class this project's own code closes stays shared rather than copied —
+ * plus lowercasing and whitespace collapse. Whitespace collapse runs before
+ * comparison, not only trimming, so `'buy crypto now'` and
+ * `'BUY   Crypto  NOW'` normalise alike.
+ *
+ * *Cost if this is too narrow:* two members posting the same text with
+ * different leetspeak spellings ('n0w' vs 'now') are not treated as
+ * duplicates. That is the safe direction for a predicate in the tier the spec
+ * says needs the full two-week observation floor — widening it later is
+ * additive, narrowing a false-positive rate after the fact is not.
+ */
+export function normaliseForDuplicateCheck(text: string): string {
+  return stripZeroWidth(text).toLowerCase().replace(/\s+/g, ' ').trim();
+}
+
+/**
+ * `duplicate_message` — the same normalised text repeated within a window
+ * (Task 4). Consumes Task 1's `createWindowStore` via `countMatching`.
+ *
+ * The store lives in `compile`'s closure, not module scope (plan ruling
+ * PR-21, `window.ts`'s own doc comment): editing this rule's config changes
+ * its fingerprint and rebuilds the artifact — and this store — from nothing,
+ * resetting the window. That is correct: a repeat count accrued under an old
+ * threshold should not be judged against a new one.
+ *
+ * The ring buffer's capacity is `maxRepeats + 1`, for the same reason
+ * `rapid_spam`'s is `maxMessages + 1`: this predicate must be able to observe
+ * one repeat more than the threshold allows, and a buffer capped at exactly
+ * the threshold could never hold that many at once.
+ */
+export const duplicateMessagePredicate: Predicate<DuplicateMessageConfig> = {
+  kind: 'duplicate_message',
+  config: duplicateMessageConfig,
+  compile(rule: AutomodRule, config: DuplicateMessageConfig): CompiledPredicate {
+    const store = createWindowStore({
+      maxEventsPerMember: config.maxRepeats + 1,
+      maxMembers: DEFAULT_MAX_TRACKED_MEMBERS,
+    });
+
+    return {
+      evaluate(input) {
+        // Only messageCreate counts, the same reasoning rapid_spam's PR-23
+        // comment gives: an edited message is not a new post.
+        if (input.eventKind !== 'messageCreate') return undefined;
+
+        const normalised = normaliseForDuplicateCheck(input.content);
+        if (normalised.length < config.minLength) return undefined;
+
+        const window = store.for({
+          // CLAUDE.md rule 2: guildId is named explicitly even though the
+          // artifact this store lives inside is already per-guild.
+          guildId: input.guildId,
+          memberId: input.authorId,
+          ruleId: rule.ruleId,
+        });
+        window.record(input.atMs, normalised);
+        const count = window.countMatching(input.atMs, config.windowMs, normalised);
+        if (count <= config.maxRepeats) return undefined;
+
+        return {
+          categoryId: 'duplicate_message',
+          // The whole message is the evidence; there is no span to point at.
+          snippet: input.content.slice(0, 200),
+          trace: {
+            kind: 'duplicate_message',
+            // The configured threshold, never message text (ruling R-8).
+            matchedOn: `more than ${config.maxRepeats} repeats within ${config.windowMs}ms`,
+            span: undefined,
+            normalisations: ['zero-width stripped', 'lowercased', 'whitespace collapsed'],
+            detail: {
+              // The repeat count and the normalised length, never the
+              // normalised text itself (Task 4): the content is a
+              // member-controlled string, and ADR 0041 §4 keeps whole-message
+              // content out of the trace.
+              count,
+              maxRepeats: config.maxRepeats,
+              windowMs: config.windowMs,
+              normalisedLength: normalised.length,
+            },
+          },
+        };
+      },
+    };
+  },
+};

+        // `countEveryoneAs` (PR-3: `mentionsEveryone` is Discord's own gateway

+import { z } from 'zod';
+import { createWindowStore, DEFAULT_MAX_TRACKED_MEMBERS } from '../window.js';
+import type { AutomodRule, CompiledPredicate, Predicate } from './types.js';
+
+export const rapidSpamConfig = z
+  .object({
+    /** Fires when a member's `messageCreate` count inside `windowMs` exceeds this. */
+    maxMessages: z.number().int().min(1).max(1_000),
+    /**
+     * How far back to count messages, in milliseconds. Bounded 1 s-10 min so
+     * a config cannot ask the window store to hold an unbounded span — the
+     * per-member ring buffer this predicate builds is sized from
+     * `maxMessages`, not from `windowMs`, but an unbounded window would still
+     * mean "everything this member has ever posted" is one config edit away.
+     */
+    windowMs: z.number().int().min(1_000).max(600_000),
+  })
+  .strict();
+
+export type RapidSpamConfig = z.infer<typeof rapidSpamConfig>;
+
+/**
+ * `rapid_spam` — messages per member per window (Task 3, plan ruling PR-22).
+ *
+ * The window store is built here, in `compile`, not at module scope: it lives
+ * in the closure of the returned `CompiledPredicate`, which is what gives it
+ * the right lifetime (plan ruling PR-21, `window.ts`'s own doc comment). **A
+ * config edit resets this rule's window** — editing `maxMessages` or
+ * `windowMs` changes the rule's content fingerprint, which rebuilds the
+ * artifact (and this store) from nothing. That is correct: a member's count
+ * accrued under an old threshold should not be judged against a new one. It
+ * reads as a bug only without this comment, which is why PR-21 insists on one.
+ *
+ * The ring buffer's capacity is `maxMessages + 1`, not `maxMessages`: this
+ * predicate must be able to *observe* one message more than the threshold
+ * allows (the "one too many" message is what fires it), and a buffer capped
+ * at exactly `maxMessages` could never hold that many at once — it would
+ * evict the oldest before the count ever read above the threshold.
+ */
+export const rapidSpamPredicate: Predicate<RapidSpamConfig> = {
+  kind: 'rapid_spam',
+  config: rapidSpamConfig,
+  compile(rule: AutomodRule, config: RapidSpamConfig): CompiledPredicate {
+    const store = createWindowStore({
+      maxEventsPerMember: config.maxMessages + 1,
+      maxMembers: DEFAULT_MAX_TRACKED_MEMBERS,
+    });
+
+    return {
+      evaluate(input) {
+        // PR-23: only `messageCreate` counts toward the rate. A member fixing
+        // a typo three times must not read as three messages — the content
+        // predicates still see the edit; this predicate alone must not.
+        if (input.eventKind !== 'messageCreate') return undefined;
+
+        const window = store.for({
+          // CLAUDE.md rule 2: guildId is named explicitly even though the
+          // artifact this store lives inside is already per-guild.
+          guildId: input.guildId,
+          memberId: input.authorId,
+          ruleId: rule.ruleId,
+        });
+        window.record(input.atMs);
+        const count = window.countWithin(input.atMs, config.windowMs);
+        if (count <= config.maxMessages) return undefined;
+
+        return {
+          categoryId: 'rapid_spam',
+          // The whole message is the evidence; there is no span to point at.
+          snippet: input.content.slice(0, 200),
+          trace: {
+            kind: 'rapid_spam',
+            // The configured threshold, never message text (ruling R-8).
+            matchedOn: `more than ${config.maxMessages} messages within ${config.windowMs}ms`,
+            // A property of a sequence of messages, not a span of this one.
+            span: undefined,
+            normalisations: [],
+            detail: {
+              count,
+              maxMessages: config.maxMessages,
+              windowMs: config.windowMs,
+            },
+          },
+        };
+      },
+    };
+  },
+};

+ * A bounded, guild-scoped sliding-window store (Release 2 Slice 4).
+ *
+ * The first state any predicate on this project has carried. Everything
+ * before this slice was pure: compile a rule, evaluate a message, done. The
+ * window tier (`rapid_spam`, `duplicate_message`) needs to remember recent
+ * events per member, which is what this module holds — nothing else does or
+ * should (plan ruling PR-21).
+ *
+ * **No `Date.now()` anywhere in this module.** Every method takes `atMs` from
+ * its caller. `runDryRun` replays fixtures at their recorded timestamps
+ * (ADR 0041 §6); a store that read the wall clock would make a replay's
+ * result depend on when the replay ran, which breaks the determinism that
+ * guarantee exists to provide. This is checked by a dedicated test below and
+ * is not negotiable.
+ *
+ * **Lifetime:** this store is not module state. It is built inside a
+ * predicate's `compile` and lives in the closure of the returned
+ * `CompiledPredicate`, which the subscriber caches per `<guildId>:<ruleId>`
+ * fingerprint (`evaluate.ts`'s `PredicateCache`, ADR 0040 §5) and a dry run
+ * builds fresh per invocation (ADR 0041 §6). Editing a windowed rule's config
+ * therefore resets its window — the fingerprint changes and the artifact
+ * (this store included) is rebuilt from nothing. That is correct: a changed
+ * threshold should not be judged against counts accrued under the old one
+ * (plan ruling PR-21).
+ */

+export interface WindowKey {
+  readonly guildId: string;
+  readonly memberId: string;
+  readonly ruleId: string;

+/** One member's recent events for one rule, oldest first. */
+export interface MemberWindow {

+   * Records an event at `atMs`, dropping the oldest if the per-member cap is

+   * window, so an out-of-window event is never counted anyway.
+   */

+  /** How many events fall inside `[atMs - windowMs, atMs]`. */
+  countWithin(atMs: number, windowMs: number): number;

+  countMatching(atMs: number, windowMs: number, value: string): number;
+}
+
+export interface WindowStore {
+  for(key: WindowKey): MemberWindow;
+  /** Members currently tracked. Exposed for the eviction test, not for callers. */

+interface Event {
+  readonly atMs: number;
+  readonly value: string | undefined;
+}

+/**
+ * A guild+member+rule's ring buffer.
+ *
+ * **There is deliberately no age-based pruning, and the reason is worth
+ * stating because an earlier draft had some.** Correctness does not come from
+ * pruning: `countWithin` and `countMatching` filter by the window on *every*
+ * call, so an event outside the window is never counted whether it is still in
+ * the array or not. And memory is bounded by `maxEventsPerMember` alone, which
+ * is applied on every `record`, so the array is capped at all times.
+ *
+ * That left pruning with **no observable effect** — which is exactly how it was
+ * found: the plan required a "watch it fail" guard for it, and the implementer
+ * reported honestly that disabling pruning changed nothing any test could see.
+ * A mechanism whose guard cannot be written is a mechanism to justify or
+ * delete, and this one could not be justified: it bought a slightly smaller
+ * steady-state array inside an already-capped buffer, at the cost of tracking
+ * the largest window ever observed so `record` had something to prune against.
+ *
+ * `onRecord` is invoked from `record()` — not from the read methods — because
+ * recording is this member's only observable "activity" for eviction
+ * purposes (see `createWindowStore`'s doc comment for why a read must not
+ * trigger registration or eviction).
+ */
+class RingMemberWindow implements MemberWindow {
+  private events: Event[] = [];
+
+  constructor(
+    private readonly maxEventsPerMember: number,
+    private readonly onRecord: () => void,
+  ) {}
+
+  record(atMs: number, value?: string): void {
+    this.onRecord();
+    this.events.push({ atMs, value });
+    // Bound 1 of 2 (PR-22): never more than maxEventsPerMember events, for
+    // this member, for this rule. This is the *only* thing bounding the array,
+    // and it runs on every record, so the array is capped at all times.
+    if (this.events.length > this.maxEventsPerMember) {
+      this.events = this.events.slice(this.events.length - this.maxEventsPerMember);
+    }
+  }
+
+  countWithin(atMs: number, windowMs: number): number {
+    const floor = atMs - windowMs;
+    return this.events.filter((e) => e.atMs >= floor && e.atMs <= atMs).length;
+  }
+
+  countMatching(atMs: number, windowMs: number, value: string): number {
+    const floor = atMs - windowMs;
+    return this.events.filter((e) => e.atMs >= floor && e.atMs <= atMs && e.value === value).length;
+  }
+}

+/**

+ *
+ * Two independent caps (plan ruling PR-22): `maxEventsPerMember` bounds one
+ * member's own ring buffer, and `maxMembers` bounds how many members are
+ * tracked at once across the whole guild, evicting the least-recently-active
+ * past that cap. Both are needed — without the per-member cap a single
+ * member's flood grows one buffer without limit; without the member cap a
+ * guild-wide raid grows the map without limit.
+ *

+ * spammer may get one extra message through where refusing to track would
+ * mean the rule silently stops working under exactly the load it exists for.
+ * That is the safe direction, and it is bounded and documented rather than
+ * silent.

+ * member who has never posted (or whose window was already evicted) must be
+ * queryable without side effects: `for(key)` on an absent key returns a

+ * mean a caller merely *reading* an evicted member's count (as `doctor` or a
+ * future admin surface might, and as this module's own eviction test does)

+ * an empty placeholder — silently destroying live state to answer a read.

+ * another member their slot.
+ */

+  // Map iteration order is insertion order, and this store always re-inserts
